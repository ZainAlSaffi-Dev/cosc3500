\newcounter{sigentries}
\newcommand{\sigstats}[1]{%
  \setcounter{sigentries}{0}%
  \foreach \sigignored in {#1}{\stepcounter{sigentries}}%
  \pgfmathsetmacro{\sigcolwidth}{(1 - 0.04*(\value{sigentries}-1))/\value{sigentries}}%
  \par\medskip\noindent
  \foreach \sigvalue/\sigcaption [count=\sigindex] in {#1}{%
    \ifnum\sigindex>1\hfill\fi
    \begin{minipage}[t]{\sigcolwidth\linewidth}
      \raggedright{\color{sigblue}\Large\bfseries \sigvalue}\par\vspace{0.15em}
      {\color{sigslate}\scriptsize \sigcaption}
    \end{minipage}%
  }%
  \par
}

% --------------------------------------------------------------------------
% 0:45 The equation
% --------------------------------------------------------------------------

\begin{frame}{The equation}{The price $V(S, v, t)$ solves one PDE, every other symbol is a known number}
  \vspace*{-0.8em}
  \begin{block}{The Heston pricing equation}
    \vspace{-0.6em}
    \begin{equation*}
      V_t
      + \underbrace{\tfrac{1}{2} v S^2 V_{SS} + \rho\xi v S\, V_{Sv} + \tfrac{1}{2}\xi^2 v\, V_{vv}}_{\text{randomness in } S \text{ and } v}
      + \underbrace{(r - q) S\, V_S + \kappa(\theta - v)\, V_v}_{\text{drift}}
      - r V = 0
    \end{equation*}
    \vspace{-1.0em}
  \end{block}
  \begin{block}{On the grid, each derivative is a difference of neighbouring cells}
    \vspace{-0.6em}
    {\small
    \begin{gather*}
      V_S \approx \frac{V_{i+1,j} - V_{i-1,j}}{2\,\Delta S}
      \qquad
      V_{SS} \approx \frac{V_{i+1,j} - 2V_{i,j} + V_{i-1,j}}{\Delta S^2}
      \qquad
      V_v,\ V_{vv} \text{ the same way in } j
      \\
      V_{Sv} \approx \frac{V_{i+1,j+1} - V_{i-1,j+1} - V_{i+1,j-1} + V_{i-1,j-1}}{4\,\Delta S\,\Delta v}
    \end{gather*}}
    \vspace{-1.2em}
    \begin{equation*}
      V^{\text{new}}_{i,j} = V_{i,j} + \Delta t \,\big(\text{the five terms} - r V_{i,j}\big)
    \end{equation*}
    \vspace{-1.2em}
  \end{block}
  \sigcallout{One explicit step reads only the old values, so every cell in a
    step can be computed at the same time.}
\end{frame}

\begin{frame}{How every number was measured}{One exclusive rangpur r730 node through sbatch, never the login node}
  \vspace*{-0.4em}
  \begin{sigcolumns}
    \begin{sigcolumn}
      {\small
      \begin{itemize}
        \item Dual Xeon E5-2670 v3, 2 x 12 cores, 30 MiB L3 per socket, g++ 8.5, Open MPI 4.1.
        \item \texttt{-{}-exclusive} on a named node, so no other job shares it.
        \item Median of 5 repetitions. Threads pinned to cores.
        \item The tests run first in the same job. The price is printed, so the compiler cannot delete the loop.
      \end{itemize}}
    \end{sigcolumn}
    \begin{sigcolumn}
      \begin{block}{A first round was thrown away}
        Short runs leave cells below $2.2 \times 10^{-308}$, which this CPU handles slowly. That varied with the grid, so rungs ran in different regimes. Every sweep was rerun with these flushed to zero.
      \end{block}
    \end{sigcolumn}
  \end{sigcolumns}
  \sigcallout{Rates are million cell updates per second on the 2048 by 512 grid.}
  \sigcite{Denormal numbers, D.\,Goldberg, What every computer scientist should know about floating-point arithmetic, \emph{ACM Computing Surveys} 23(1), 5--48, 1991.}
\end{frame}
